function openClaude(pid){ var webUrl='https://claude.ai/project/'+pid; var deepLink='claude://project/'+pid; // On iOS Safari, window.location.href is the only reliable way to trigger // a custom URI scheme. The iframe approach doesn't work in mobile Safari. // We set the deep link, then fall back to the web URL after 1.5s if the // app doesn't intercept (i.e. app isn't installed or scheme isn't registered). var fallback=setTimeout(function(){window.open(webUrl,'_blank');},1500); window.addEventListener('blur',function onBlur(){ // App opened — cancel the web fallback clearTimeout(fallback); window.removeEventListener('blur',onBlur); }); window.location.href=deepLink; }
